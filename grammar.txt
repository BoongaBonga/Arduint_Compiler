Statement = 
  | FunctionDeclaration
  | DefineStatement
  | loopStatement
  | ifStatement
  | PrintStatement

BlockStatement =
    '{' + Statement + '}'
  | FunctionDeclaration + BlockStatement

--------------------------------------------------------

Expression:
  Assignment

Assignment:
  NAME '=' Expression
  | LogicalOr

LogicalOr:
  LogicalOr '|' LogicalAnd
  | LogicalAnd

LogicalAnd:
  LogicalAnd 'and' Comparison  
  | Comparison

Comparison:
  Comparison '==' BitwiseOr 
  | BitwiseOr

BitwiseOr:
  BitwiseOr '|' BitwiseXor
  | BitwiseXor

BitwiseXor:
  BitwiseXor 'xor' BitwiseAnd
  | BitwiseAnd

BitwiseAnd:
  BitwiseAnd '&' Sum    
  | Sum

Sum:
  Sum '+' Factor
  | Factor

Factor:
  Factor '*' Unary
  | Unary

Unary:
  '-' Unary
  | '!' Unary
  | Power

Power: 
  PostFix '^' Power
  | PostFix

PostFix:
  PostFix '++'
  | PostFix '--'
  | Grouping

Grouping:
  '(' Expression ')'
  | Atom

Atom:
  VALUE
  | NAME
